import json

from pathlib import Path

import golden
from golden import problems as problem_bank
from golden.eval.misconception import T2_MATCHER_VERSION, match_misconception
from golden.sim.errors import MISCONCEPTIONS_LF_PATH

# 실행 코드 출처 — 지금 폴더(main 사본의 src/backend)의 golden이어야 한다(다른 사본 코드가 돌면 거부)

PROBLEMS = ("LF-01", "LF-02", "LF-03", "LF-04", "LF-05")

run_dir, out_path = Path(sys.argv[1]), Path(sys.argv[2])

seeds = load_seed_utterances()
bank = {p["id"]: p for p in problem_bank.load_problems()}
bundles = injection_bundles()
mis_ops = {
    str(r["mis_id"]): tuple(r.get("op_code") or ())
    for r in json.loads(MISCONCEPTIONS_LF_PATH.read_text(encoding="utf-8"))["misconceptions"]

# 회차 = 그 페르소나의 몇 번째 블록인가(러너는 회차 → 페르소나 → 문항 순으로 한 줄씩 쓴다)
blocks: dict[str, int] = {}
prev = None

for row in rows:
    p = row.get("persona")
    if p != prev:
        blocks[p] = blocks.get(p, 0) + 1
        prev = p
    if p in PERSONAS and blocks[p] == 1 and row.get("problem_id") in PROBLEMS:
        key = (p, row["problem_id"])
        if key in picked:

problems = []
if meta.get("mode") != "live":
    problems.append(f"mode={meta.get('mode')}(live여야 한다)")
if t2_flag != "1":

if missing:

if bad_http:
    problems.append(f"http_status 2xx 아님 {bad_http}")
fake = [f"{k[0]}/{k[1]}" for k, r in picked.items() if any(str(t.get("tutor", "")).startswith(FAKE_PREFIX) for t in r.get("turns", []) if t.get("kind") == "tutor_turn")]

# 숫자 칸(세션이 다시 세어 대조한다)
num = {"wrong": 0, "wrong_matched": 0, "tt_wrong": 0, "tt_matched": 0, "tt_matched_next_correct": 0,
       "tt_unmatched": 0, "tt_unmatched_next_correct": 0, "sess_matched": 0, "sess_matched_correct": 0,
       "p3_matched_turns": 0, "p3_same_op": 0, "p3_in_bundle": 0}
masked = 0
lines = []
for p in PERSONAS:
    for q in PROBLEMS:
        r = picked.get((p, q))
        if r is None:
            continue
        turns = r.get("turns", [])
        last = turns[-1] if turns else {}
        op = r.get("injected_op")
        bundle = bundles.get((q, str(r.get("forced_answer")))) if r.get("forced_answer") is not None else None
        lines.append(

        any_match = False
        for i, t in enumerate(turns):

                        num["tt_matched"] += 1
                        num["tt_matched_next_correct"] += nxt_ok
                    else:
                        num["tt_unmatched"] += 1
                        num["tt_unmatched_next_correct"] += nxt_ok
                if mis is not None:
                    any_match = True
                    if p == "P3":

    f"matcher={T2_MATCHER_VERSION} personas_in_run={','.join(meta.get('personas') or [])}"

nums = "S29_NUM " + " ".join(f"{k}={v}" for k, v in num.items())

print(head)
print(nums)
print(f"OUT_FILE={out_path} CHARS={len(out_path.read_text(encoding='utf-8'))}")

- 클립보드 글(`%TEMP%\s29_15.txt`)에는 고른 15건의 학생·튜터 발화와 숫자 줄이 들어 있다. 붙여넣은 뒤 그 파일은 지워도 된다(`Remove-Item $env:TEMP\s29_15.txt`).
- 시드를 베낀 학생 발화는 학생 모델 단계에서 이미 대본 문장으로 바뀐다(`golden/sim/student.py` `copies_seed`). 이 스크립트가 같은 검사를 한 번 더 해 `[시드 일치 — 가림]`으로 바꾸고 건수(`masked=`)를 낸다 — 공통 규칙 7(시드는 로컬 Ollama 밖으로 보내지 않는다). 매칭은 가리기 전 원문으로 하지만, 내보내는 것은 오개념 ID뿐이다.
- 고르는 기준은 페르소나·문항·1회차 셋뿐이다. 1회차 = 기록 파일에서 그 페르소나의 첫 블록(러너가 회차 → 페르소나 → 문항 순으로 한 줄씩 쓴다 · `golden/sim/run.py` `run`). S24 추출기와 같은 규칙이다.
- T2 매칭 줄(`T2: 매칭=… · 힌트=…`)은 실행 때 서버가 부른 것과 같은 순수 함수(`golden.eval.misconception.match_misconception` · `t2-match-v1`)를 같은 입력에 다시 불러 만든다. DB(`golden.eval_log`)를 읽지 않는 이유: 시연용 DB는 일회용이라 지금 남아 있다고 가정할 수 없다. 힌트 여부는 api 규칙 그대로다(플래그 1 · 튜터 턴 · 매칭 있음 → 힌트 넣음 · 판정 턴은 튜터를 부르지 않으므로 늘 아니오).
- `S29_NUM` 줄 칸: `wrong`(틀린 입력 수 — 튜터 턴 + 오답 판정 턴) · `wrong_matched` · `tt_wrong`(튜터 턴만) · `tt_matched` · `tt_matched_next_correct`(매칭된 튜터 턴 바로 다음 입력이 정답) · `tt_unmatched` · `tt_unmatched_next_correct` · `sess_matched`(매칭이 한 번이라도 있던 세션) · `sess_matched_correct`(그중 정답으로 끝난 세션) · `p3_matched_turns` · `p3_same_op`(매칭된 오개념의 OP 코드에 주입 OP가 들어 있다) · `p3_in_bundle`(매칭된 오개념이 주입 답의 오개념 묶음 안에 있다). 세션은 글을 받아 손으로 다시 세어 대조한다.

## 클라우드 시험(이 런북을 쓰기 전)

합성 실행 기록(회차 2 · 페르소나 4 · 회차마다 문항 순서 섞음 · P3는 주입기 답)으로 스크립트를 돌렸다: 정상 → `ok=True picked=15` · 시드 베낌 1건 → `masked=1` 통과 · 가짜 모드 튜터 발화 → 거부 · 조합 하나 빠짐 → `picked=14` 거부 · http 2xx 아님 → 거부 · `t2_flag=0` 실행 → 거부 · 다른 위치의 golden 코드 → `S29_REFUSED` 거부. 정상 기록에서 P3 주입 답은 20턴 모두 매칭되고 계열이 같았다(`p3_same_op=20 p3_in_bundle=20`) — 주입기와 매칭표가 같은 규칙에서 나오므로 그래야 한다.
