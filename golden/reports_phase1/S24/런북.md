## 블록 (창 A)

```powershell
# [창 A · S24 15건 추출] Windows PowerShell — 읽기 전용(저장소·실행 기록을 바꾸지 않는다)
cd C:\Users\kiki\Desktop\__AI\WhyMath
$Root = "C:\Users\kiki\Desktop\__AI"
$Main = "$Root\WhyMath"
$RunDir = "$Root\sim-s23\out\after"
$Py = "$Main\src\backend\.venv\Scripts\python.exe"
$SeedDir = "$Main\golden\sim\seed"
$Wt = Join-Path $env:TEMP "whymath-s24"
$Out = Join-Path $env:TEMP "s24_15.txt"
$Script = Join-Path $env:TEMP "extract_s24.py"
$ExtractExit = $null
git -C $Main fetch -q origin main
git -C $Main worktree prune
if (Test-Path -LiteralPath $Wt) { git -C $Main worktree remove --force $Wt }
git -C $Main worktree add -q --detach $Wt origin/main
"WT_HEAD=$(git -C $Wt log -1 --format=%h)"
@'
# S24 추출기 — S23 after 실행 기록에서 P2·P3·P4 × LF-01~05 × 1회차 15건을 골라 세션에 넘길 글로 만든다
# 사용: python extract_s24.py <실행 폴더> <출력 txt>   (cwd = 저장소 src/backend · GOLDEN_SIM_SEED_DIR 설정)
# 고르는 기준은 셋뿐(페르소나 · 문항 · 1회차) — 내용으로 고르지 않는다(지시서 S24 할 일 2)
import json
import sys
from pathlib import Path

from golden.sim.student import copies_seed, load_seed_utterances
from golden.tutor import FAKE_PREFIX
import golden

# 실행 코드 출처 — 지금 폴더(main 사본의 src/backend)의 golden이어야 한다(다른 사본 코드가 돌면 거부)
if not Path(golden.__file__).resolve().is_relative_to(Path.cwd().resolve()):
    sys.exit(f"S24_REFUSED — golden이 다른 곳에서 왔다: {golden.__file__}")
print(f"GOLDEN_FROM={Path(golden.__file__).resolve().parent}")

PERSONAS = ("P2", "P3", "P4")
PROBLEMS = ("LF-01", "LF-02", "LF-03", "LF-04", "LF-05")

run_dir, out_path = Path(sys.argv[1]), Path(sys.argv[2])
meta = json.loads((run_dir / "run_meta.json").read_text(encoding="utf-8"))
rows = [json.loads(ln) for ln in (run_dir / "sessions.jsonl").read_text(encoding="utf-8").splitlines() if ln.strip()]
seeds = load_seed_utterances()

# 회차 = 그 페르소나의 몇 번째 블록인가(러너는 회차 → 페르소나 → 문항 순으로 한 줄씩 쓴다)
blocks: dict[str, int] = {}
prev = None
picked = {}
for row in rows:
    p = row.get("persona")
    if p != prev:
        blocks[p] = blocks.get(p, 0) + 1
        prev = p
    if p in PERSONAS and blocks[p] == 1 and row.get("problem_id") in PROBLEMS:
        key = (p, row["problem_id"])
        if key in picked:
            sys.exit(f"S24_REFUSED — 1회차에 같은 문항이 두 번 나왔다: {key}")
        picked[key] = row

problems = []
if meta.get("mode") != "live":
    problems.append(f"mode={meta.get('mode')}(live여야 한다)")
missing = [f"{p}/{q}" for p in PERSONAS for q in PROBLEMS if (p, q) not in picked]
if missing:
    problems.append(f"빠진 조합 {missing}")
bad_http = [f"{k[0]}/{k[1]}={r.get('http_status')}" for k, r in picked.items() if not (isinstance(r.get("http_status"), int) and 200 <= r["http_status"] < 300)]
if bad_http:
    problems.append(f"http_status 2xx 아님 {bad_http}")
fake = [f"{k[0]}/{k[1]}" for k, r in picked.items() if any(str(t.get("tutor", "")).startswith(FAKE_PREFIX) for t in r.get("turns", []) if t.get("kind") == "tutor_turn")]
if fake:
    problems.append(f"가짜 모드 튜터 발화 {fake}")

masked = 0
lines = []
for p in PERSONAS:
    for q in PROBLEMS:
        r = picked.get((p, q))
        if r is None:
            continue
        turns = r.get("turns", [])
        last = turns[-1] if turns else {}
        lines.append(
            f"## {p} · {q} · injected_op={r.get('injected_op')} · elapsed_s={r.get('elapsed_s')} · "
            f"http={r.get('http_status')} · 턴수={len(turns)} · 마지막={last.get('kind')} · correct={last.get('correct')}"
        )
        for t in turns:
            s = str(t.get("student", ""))
            if copies_seed(s, seeds):
                s = "[시드 일치 — 가림]"
                masked += 1
            lines.append(f"[{t.get('n')}·{t.get('kind')}] 학생: {s}")
            lines.append(f"[{t.get('n')}·{t.get('kind')}] 튜터: {t.get('tutor', '')}")
        lines.append("")

ok = not problems
head = (
    f"S24_CHECK ok={ok} picked={len(picked)} masked={masked} mode={meta.get('mode')} "
    f"personas_in_run={','.join(meta.get('personas') or [])}"
)
if problems:
    head += " problems=" + " | ".join(problems)
out_path.write_text(head + "\n\n" + "\n".join(lines), encoding="utf-8")
print(head)
print(f"OUT_FILE={out_path} CHARS={len(out_path.read_text(encoding='utf-8'))}")
sys.exit(0 if ok else 1)
'@ | Set-Content -Path $Script -Encoding UTF8
$Ready = (Test-Path "$RunDir\sessions.jsonl") -and (Test-Path "$RunDir\run_meta.json") -and (Test-Path $Py) -and (Test-Path "$SeedDir\utterances.txt") -and (Test-Path "$Wt\src\backend\golden\sim\student.py") -and (Test-Path $Script)
"READY=$Ready"
if ($Ready) { $env:PYTHONPATH = "$Wt\src\backend"; $env:GOLDEN_SIM_SEED_DIR = $SeedDir; Set-Location "$Wt\src\backend"; & $Py $Script $RunDir $Out; $ExtractExit = $LASTEXITCODE; "EXTRACT_EXIT=$ExtractExit" } else { "EXTRACT_SKIPPED — READY=False: sessions.jsonl·run_meta.json(sim-s23\out\after) · .venv python · 시드 utterances.txt · main 사본 중 없는 것이 있다" }
if ($ExtractExit -eq 0) { Get-Content -Path $Out -Raw -Encoding UTF8 | Set-Clipboard; "CLIPBOARD=True — 세션 대화창에 Ctrl+V로 붙여넣어 보내기" } else { "CLIPBOARD=False — 위 S24_CHECK 줄(또는 EXTRACT_SKIPPED 줄)만 세션에 보내기" }
Set-Location $Main
git -C $Main worktree remove --force $Wt
Remove-Item -Path $Script -Force -ErrorAction SilentlyContinue
```

- 클립보드 글(`%TEMP%\s24_15.txt`)에는 고른 15건의 학생·튜터 발화가 들어 있다. 붙여넣은 뒤 그 파일은 지워도 된다(`Remove-Item $env:TEMP\s24_15.txt`).
- 시드를 베낀 학생 발화는 학생 모델 단계에서 이미 대본 문장으로 바뀐다(`golden/sim/student.py` `copies_seed`). 이 스크립트가 같은 검사를 한 번 더 해 `[시드 일치 — 가림]`으로 바꾸고 건수(`masked=`)를 낸다 — 공통 규칙 7(시드는 로컬 Ollama 밖으로 보내지 않는다).
- 고르는 기준은 페르소나·문항·1회차 셋뿐이다. 1회차 = 기록 파일에서 그 페르소나의 첫 블록(러너가 회차 → 페르소나 → 문항 순으로 한 줄씩 쓴다 · `golden/sim/run.py` `run`).

## 클라우드 시험(이 런북을 쓰기 전)

합성 실행 기록(회차 2 · 페르소나 4 · 회차마다 문항 순서 섞음)으로 스크립트를 돌렸다: 정상 → `ok=True picked=15`(전부 1회차) · 시드 베낌 1건 → `masked=1` 통과 · 가짜 모드 튜터 발화 → 거부 · 조합 하나 빠짐 → `picked=14` 거부 · http 2xx 아님 → 거부 · 다른 위치의 golden 코드 → `S24_REFUSED` 거부.
