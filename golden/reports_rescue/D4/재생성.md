- 아래 코드를 저장소 밖에 `build_ledger.py`로 저장하고 저장소 루트에서 `python3 /tmp/build_ledger.py . golden/reports_rescue/D4`를 실행한다.

```python
"""RS-D4 스크립트 대장 생성기 — 파트 F(scripts/ · src/data-pipeline/)의 .py마다 "부르는 곳"을 판정한다.

실행하지 않는다: 스크립트를 import·실행하지 않고, ast 파싱과 문자열 검색만 한다.
사용: python3 build_ledger.py <저장소 루트> <출력 폴더>
"""

from __future__ import annotations

import ast
import collections
import re
import subprocess
import sys
from pathlib import Path

ROOT = Path(sys.argv[1]).resolve()
OUT = Path(sys.argv[2]).resolve()

# 검색 대상에서 빼는 경로 — 학생 데이터(규칙 15)와 RS 묶음 자신의 산출물(자기 참조 방지)
EXCLUDE_PREFIXES = (
    "golden/logs/",
    "golden/sim/seed/",
    "golden/reports_rescue/",
)

# 부르는 곳 4범주 — 위에서부터 우선한다
CATEGORY_DIRS = {
    "CI": (".github/workflows/",),
    "하네스·훅": (".claude/", "scripts/harness/"),
    "문서만": ("docs/", "golden/", "manuals/"),
}
# 텍스트로 읽을 확장자 — 바이너리(이미지·xlsx·pdf)는 뺀다
TEXT_SUFFIXES = {
    ".md", ".yml", ".yaml", ".json", ".jsonl", ".py", ".sh", ".ps1", ".psm1", ".bat", ".cmd",
    ".txt", ".toml", ".cfg", ".ini", ".csv", ".tsv", ".html", ".dart", ".ts", ".tsx", ".js", "",
}


def git_ls(*patterns: str) -> list[str]:
    """추적 중인 파일만 나열한다(.gitignore 대상·로그 제외)."""
    out = subprocess.run(
        ["git", "ls-files", "-z", *patterns], cwd=ROOT, capture_output=True, check=True
    ).stdout.decode("utf-8")
    return [p for p in out.split("\0") if p]


def read_text(rel: str) -> str:
    try:
        return (ROOT / rel).read_text(encoding="utf-8", errors="replace")
    except (OSError, UnicodeError) as exc:  # 읽기 실패는 숨기지 않고 표시
        print(f"[읽기 실패] {rel}: {type(exc).__name__}", file=sys.stderr)
        return ""


def module_name(rel: str) -> str:
    """파일 경로 → 점 표기 모듈 이름(python -m 형태)."""
    p = rel[:-3]
    if p.startswith("src/data-pipeline/"):
        p = p[len("src/data-pipeline/"):]
    if p.endswith(("/__init__", "/__main__")):
        p = p.rsplit("/", 1)[0]
    return p.replace("/", ".")


def legacy_imports(rel: str) -> tuple[list[str], str]:
    """ast로 import를 읽어 옛 최상위 패키지(whymath_backend.<하위>)·data_pipeline·golden을 모은다."""
    src = read_text(rel)
    try:
        tree = ast.parse(src)
    except SyntaxError as exc:
        return [], f"ast 실패({type(exc).__name__})"
    found: set[str] = set()
    for node in ast.walk(tree):
        names: list[str] = []
        if isinstance(node, ast.Import):
            names = [a.name for a in node.names]
        elif isinstance(node, ast.ImportFrom) and node.module and node.level == 0:
            names = [node.module]
            if node.module == "whymath_backend":  # from whymath_backend import l1
                names = [f"whymath_backend.{a.name}" for a in node.names]
        for n in names:
            parts = n.split(".")
            if parts[0] == "whymath_backend":
                found.add(".".join(parts[:2]))
            elif parts[0] in ("data_pipeline", "golden"):
                found.add(parts[0])
    return sorted(found), ""


def resolve_imports(src_rel: str, target_set: set[str]) -> set[str]:
    """한 파일의 import를 F파트 파일 경로로 풀어 돌려준다(형제·상대·data_pipeline·scripts 기준)."""
    try:
        tree = ast.parse(read_text(src_rel))
    except SyntaxError:
        return set()
    here = Path(src_rel).parent
    roots = [here, Path("scripts"), Path("scripts/harness"), Path("src/data-pipeline")]
    out: set[str] = set()

    def try_mod(base_dirs: list[Path], dotted: str) -> None:
        segs = [x for x in dotted.split(".") if x]
        if not segs:
            return
        for b in base_dirs:
            cand = b.joinpath(*segs)
            for c in (cand.with_suffix(".py"), cand / "__init__.py"):
                cs = c.as_posix()
                if cs in target_set:
                    out.add(cs)
                    return

    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            for a in node.names:
                try_mod(roots, a.name)
        elif isinstance(node, ast.ImportFrom):
            if node.level:
                base = here
                for _ in range(node.level - 1):
                    base = base.parent
                bases = [base]
            else:
                bases = roots
            modname = node.module or ""
            if modname:
                try_mod(bases, modname)
            for a in node.names:  # from 패키지 import 하위모듈
                try_mod(bases, f"{modname}.{a.name}" if modname else a.name)
    out.discard(src_rel)
    return out


def main() -> None:
    targets = sorted(git_ls("scripts/*.py") + git_ls("src/data-pipeline/*.py"))
    # 검색 말뭉치: 범주 폴더 + tests/(참고 열) + src/backend(참고 열) + scripts·data-pipeline(상호 import 참고)
    corpus_files = [
        f
        for f in git_ls(".github/workflows", ".claude", "scripts", "docs", "golden", "manuals", "tests", "src/backend", "src/data-pipeline")
        if not f.startswith(EXCLUDE_PREFIXES) and Path(f).suffix in TEXT_SUFFIXES
    ]
    corpus = {f: read_text(f) for f in corpus_files}
    # import 그래프: F파트·tests·백엔드·.claude의 .py가 F파트 파일을 import하는가
    target_set = set(targets)
    importers: dict[str, set[str]] = collections.defaultdict(set)
    for f in corpus_files:
        if f.endswith(".py"):
            for dst in resolve_imports(f, target_set):
                importers[dst].add(f)
    # 색인: 파일마다 경로 모양 낱말을 한 번만 뽑는다(정규식 233회 × 수천 파일을 피함)
    tok_rx = re.compile(r"[A-Za-z0-9_./\\-]+")
    by_last: dict[str, set[tuple[str, str]]] = collections.defaultdict(set)  # 파일명 → (파일, 낱말 전체)
    by_dotted: dict[str, set[str]] = collections.defaultdict(set)  # 점 표기 접두 → 파일들
    by_dotted_exact: dict[str, set[str]] = collections.defaultdict(set)  # 점 표기 낱말 전체 → 파일들
    for f, text in corpus.items():
        for tok in set(tok_rx.findall(text)):
            tok = tok.replace("\\", "/").rstrip("/.")
            parts = tok.split("/")
            by_last[parts[-1]].add((f, tok.lstrip("./") if tok.startswith("./") else tok))
            if "/" not in tok and "." in tok:
                segs = tok.strip(".").split(".")
                by_dotted_exact[".".join(segs)].add(f)
                for i in range(2, len(segs) + 1):
                    by_dotted[".".join(segs[:i])].add(f)
    # 저장소 전체(.py)에서 파일명이 겹치는지 — 겹치면 맨 파일명 언급은 어느 파일인지 알 수 없다
    all_py_base = collections.Counter(Path(p).name for p in git_ls("*.py"))
    rows = []
    f_edges: dict[str, set[str]] = {}
    doc_hits_per_file: collections.Counter[str] = collections.Counter()
    for t in targets:
        base = Path(t).name
        mod = module_name(t)
        ambiguous = base == "__init__.py" or all_py_base[base] > 1
        def path_consistent(tok: str) -> bool:
            """낱말이 이 파일의 경로와 맞는가 — 경로 꼬리(harness/backlog.py)이거나 절대경로가 이 파일로 끝난다."""
            if tok == base:
                return not ambiguous  # 맨 파일명은 저장소에서 유일할 때만 인정
            if t.endswith("/" + tok) or t == tok:
                return tok.count("/") >= 1
            return tok.endswith("/" + t)
        matched = {f for f, tok in by_last.get(base, set()) if path_consistent(tok)}
        rule = "경로꼬리만(동명 파일 있음)" if ambiguous else "파일명"
        # 점 표기 모듈(python -m · import) — 두 마디 이상일 때만(한 마디는 오탐이 많다)
        if mod.count(".") >= 1:
            if base == "__main__.py":  # python -m 패키지 — 하위 모듈 언급과 섞이지 않게 정확 일치만
                matched |= by_dotted_exact.get(mod, set())
                rule += "+패키지실행표기"
            else:
                matched |= by_dotted.get(mod, set())
                rule += "+모듈표기"
        # ast import 경유(형제 파일 import · 상대 import) — 부른 파일의 위치로 범주를 정한다
        matched |= importers.get(t, set())
        # F파트 안의 다른 파일이 이름으로 부르거나(subprocess) import하는 경우 — 간접 판정의 간선
        f_edges[t] = {f for f in matched if f in target_set and f != t}
        hits: dict[str, list[str]] = {"CI": [], "하네스·훅": [], "문서만": [], "tests": [], "backend": [], "F파트": []}
        for f in sorted(matched):
            if f == t:
                continue
            placed = False
            for cat, prefixes in CATEGORY_DIRS.items():
                if f.startswith(prefixes):
                    hits[cat].append(f)
                    placed = True
                    if cat == "문서만":
                        doc_hits_per_file[f] += 1
                    break
            if placed:
                continue
            if f.startswith("tests/"):
                hits["tests"].append(f)
            elif f.startswith("src/backend/"):
                hits["backend"].append(f)
            elif f.startswith(("scripts/", "src/data-pipeline/")):
                hits["F파트"].append(f)
        caller = next((c for c in ("CI", "하네스·훅", "문서만") if hits[c]), "없음")
        pkgs, err = legacy_imports(t)
        lines = read_text(t).count("\n")
        alive = "살아 있음" if t.startswith(("scripts/golden_prompts/", "scripts/harness/")) else ""
        sub = t.split("/")
        if t.startswith("scripts/"):
            folder = "scripts/" + sub[1] if len(sub) > 2 else "scripts/(루트)"
        else:
            folder = "data_pipeline/" + sub[3] if len(sub) > 4 else "data_pipeline/(루트)"
        rows.append(
            {
                "경로": t,
                "하위 폴더": folder,
                "줄 수": lines,
                "옛 최상위 패키지": ";".join(pkgs) if pkgs else "-",
                "부르는 곳": caller,
                "골든 묶음 표시": alive,
                "CI 건수": len(hits["CI"]),
                "하네스·훅 건수": len(hits["하네스·훅"]),
                "문서 건수": len(hits["문서만"]),
                "판정 근거 예": (hits[caller][0] if caller != "없음" else "-"),
                "테스트 언급 수": len(hits["tests"]),
                "백엔드 언급 수": len(hits["backend"]),
                "F파트 내 언급 수": len(hits["F파트"]),
                "판정 규칙": rule,
                "비고": err,
            }
        )
    # 간접 판정: 이 파일을 (전이적으로) import하거나 이름으로 부르는 F파트 파일의 직접 판정 중 가장 강한 것.
    # 부르는 곳이 "없음"이어도 CI가 부르는 스크립트가 import하면 CI에서 실행된다.
    # 테스트(tests/)가 import하는 것은 여기서 세지 않는다 — "테스트 언급 수" 열이 따로 있다.
    rank = {"CI": 0, "하네스·훅": 1, "문서만": 2, "없음": 3}
    direct = {r["경로"]: r["부르는 곳"] for r in rows}
    for r in rows:
        best, seen, stack = "없음", {r["경로"]}, list(f_edges[r["경로"]] | importers.get(r["경로"], set()))
        while stack:
            f = stack.pop()
            if f in seen:
                continue
            seen.add(f)
            if f in direct:
                if rank[direct[f]] < rank[best]:
                    best = direct[f]
                stack.extend(f_edges[f] | importers.get(f, set()))
            elif f.startswith(".claude/") and rank[best] > 1:
                best = "하네스·훅"
        r["간접(F파트 경유)"] = best
        r["종합"] = min(r["부르는 곳"], best, key=lambda c: rank[c])
        r["F파트 안에서 import하는 파일 수"] = len(
            [f for f in importers.get(r["경로"], set()) if f in direct]
        )

    # 패키지 표식 __init__.py는 그 패키지의 어떤 모듈이 실행돼도 함께 실행된다 → 하위 모듈의 종합을 물려받는다
    for r in rows:
        if r["경로"].endswith("/__init__.py"):
            pkg_dir = r["경로"][: -len("__init__.py")]
            subs = [o["종합"] for o in rows if o["경로"].startswith(pkg_dir) and o is not r]
            inherited = min(subs + [r["종합"]], key=lambda c: rank[c])
            if inherited != r["종합"]:
                r["종합"] = inherited
                r["비고"] = (r["비고"] + " " if r["비고"] else "") + "종합=하위 모듈에서 상속"

    OUT.mkdir(parents=True, exist_ok=True)
    cols = list(rows[0].keys())
    with (OUT / "스크립트_대장.tsv").open("w", encoding="utf-8-sig", newline="\n") as fh:
        fh.write("\t".join(cols) + "\n")
        for r in rows:
            fh.write("\t".join(str(r[c]) for c in cols) + "\n")
    # 문서 한 장이 대상 이름을 몇 개 품는지 — 전수 목록(인벤토리) 문서가 "문서만"을 부풀리는지 점검용
    with (OUT / "문서별_언급수.tsv").open("w", encoding="utf-8") as fh:
        for f, n in doc_hits_per_file.most_common():
            fh.write(f"{n}\t{f}\n")
    print(f"행 {len(rows)} · 검색 말뭉치 {len(corpus)}파일")


if __name__ == "__main__":
    main()
```
