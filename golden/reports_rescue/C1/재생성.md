## agg_l1.py

```python
"""C1 집계 — l1 하위 패키지별로 D1 등급·D5 테스트 수·D2 규칙의 조용한 except를 센다.

본문을 사람이 읽지 않는다. ast로 except 블록만 판정한다(D2 보고 76행 규칙).
"""
import ast
import collections
import csv
import sys
from pathlib import Path

ROOT = Path(sys.argv[1])  # 저장소 루트
L1 = ROOT / "src/backend/whymath_backend/l1"
RR = ROOT / "golden/reports_rescue"


def sub_of(mod: str) -> str:
    # whymath_backend.l1.<sub>.<...> → <sub> · 단독 파일은 '(단독) 이름'
    parts = mod.split(".")
    if len(parts) == 2:
        return "(l1 __init__)"
    if (L1 / parts[2]).is_dir():
        return parts[2]
    return "(단독) " + parts[2]


EMPTY_CALLS = {"list", "dict", "set", "tuple", "frozenset"}


def quiet_stmt(s: ast.stmt) -> bool:
    # D2 규칙: pass·continue·...·문자열 상수·return 기본값만 허용
    if isinstance(s, (ast.Pass, ast.Continue)):
        return True
    if isinstance(s, ast.Expr) and isinstance(s.value, ast.Constant):
        return True
    if isinstance(s, ast.Return):
        v = s.value
        if v is None or isinstance(v, ast.Constant):
            return True
        if isinstance(v, (ast.List, ast.Dict, ast.Tuple, ast.Set)) and not (
            getattr(v, "elts", None) or getattr(v, "keys", None)
        ):
            return True
        if (
            isinstance(v, ast.Call)
            and isinstance(v.func, ast.Name)
            and v.func.id in EMPTY_CALLS
            and not v.args
            and not v.keywords
        ):
            return True
    return False


def broad(h: ast.ExceptHandler) -> bool:
    t = h.type
    if t is None:
        return True
    names = []
    for n in t.elts if isinstance(t, ast.Tuple) else [t]:
        if isinstance(n, ast.Name):
            names.append(n.id)
        elif isinstance(n, ast.Attribute):
            names.append(n.attr)
    return any(x in ("Exception", "BaseException") for x in names)


agg = collections.defaultdict(lambda: collections.Counter())
quiet_sites = []
for p in sorted(L1.rglob("*.py")):
    rel = p.relative_to(ROOT / "src/backend").with_suffix("")
    mod = ".".join(rel.parts)
    if mod.endswith(".__init__"):
        mod = mod[: -len(".__init__")]
    sub = sub_of(mod)
    tree = ast.parse(p.read_text(encoding="utf-8"))
    for n in ast.walk(tree):
        if isinstance(n, ast.ExceptHandler):
            agg[sub]["except"] += 1
            b = broad(n)
            agg[sub]["except_broad"] += b
            if all(quiet_stmt(s) for s in n.body):
                agg[sub]["quiet"] += 1
                agg[sub]["quiet_broad"] += b
                quiet_sites.append(f"{p.relative_to(ROOT)}:{n.lineno}{' (광범위)' if b else ''}")

# D1 대장 — 옛앱 등급 분포·import하는 모듈 수 합
with open(RR / "D1/ledger.csv", encoding="utf-8-sig") as f:
    for r in csv.DictReader(f):
        m = r["모듈"]
        if not (m == "whymath_backend.l1" or m.startswith("whymath_backend.l1.")):
            continue
        s = sub_of(m)
        agg[s]["mods"] += 1
        agg[s]["lines"] += int(r["줄 수"])
        agg[s]["legacy:" + r["옛앱 등급"]] += 1
        agg[s]["golden:" + r["골든 등급"]] += 1
        agg[s]["importers"] += int(r["이 모듈을 import하는 모듈 수"] or 0)
        if r["비고"]:
            agg[s]["note:" + r["비고"]] += 1

# D5 — 직접 import 테스트 파일 수(5번째 칸) 합·0개 모듈 수
with open(RR / "D5/모듈별_테스트수.tsv", encoding="utf-8-sig") as f:
    rd = csv.reader(f, delimiter="\t")
    next(rd)
    for r in rd:
        if r[3] != "whymath_backend.l1":
            continue
        s = sub_of(r[0])
        n = int(r[4])
        agg[s]["tests_sum"] += n
        agg[s]["tests_zero_mods"] += n == 0

order = sorted(agg)
tot = collections.Counter()
for s in order:
    c = agg[s]
    tot.update(c)
    leg = "·".join(f"{k[7:]}{v}" for k, v in sorted(c.items()) if k.startswith("legacy:"))
    gol = "·".join(f"{k[7:]}{v}" for k, v in sorted(c.items()) if k.startswith("golden:"))
    notes = "·".join(f"{k[5:]}{v}" for k, v in c.items() if k.startswith("note:"))
    print(
        f"{s}\t{c['mods']}\t{c['lines']}\t옛앱[{leg}]\t골든[{gol}]\t"
        f"imp{c['importers']}\tT{c['tests_sum']}(0개{c['tests_zero_mods']})\t"
        f"exc{c['except']}/광{c['except_broad']}\t조용{c['quiet']}/광{c['quiet_broad']}\t{notes}"
    )
print(
    f"합계\t{tot['mods']}\t{tot['lines']}\texc{tot['except']}/광{tot['except_broad']}"
    f"\t조용{tot['quiet']}/광{tot['quiet_broad']}\tT{tot['tests_sum']}(0개{tot['tests_zero_mods']})"
)
print("-- 조용한 except 위치")
for q in quiet_sites:
    print(q)
```

## count_data.py

```python
"""C1 데이터 건수(짧은 판) — 본문은 출력하지 않고 크기·레코드 수만 낸다. 인자 = 저장소 루트 + 경로들."""
import json, sys, yaml  # noqa: E401
from pathlib import Path

for rel in sys.argv[2:]:
    p = Path(sys.argv[1]) / rel
    if not p.exists():
        print(f"{rel}\t없음"); continue  # noqa: E702
    if p.is_dir():
        print(f"{rel}\t폴더 파일 {sum(f.is_file() for f in p.rglob('*'))}"); continue  # noqa: E702
    t = p.read_text(encoding="utf-8")
    o = [x for x in t.splitlines() if x.strip()] if p.suffix == ".jsonl" else (json.loads(t) if p.suffix == ".json" else yaml.safe_load(t))
    if isinstance(o, dict):  # 딕트면 리스트·딕트 값 중 긴 것 3개의 길이
        top = sorted(((k, len(v)) for k, v in o.items() if isinstance(v, (list, dict))), key=lambda x: -x[1])[:3]
        d = f"dict 키{len(o)} [" + " ".join(f"{k}={n}" for k, n in top) + "]"
    else:
        d = (f"줄 {len(o)}" if p.suffix == ".jsonl" else f"list {len(o)}")
    print(f"{rel}\t{p.stat().st_size // 1024}KB\t{d}")
```
