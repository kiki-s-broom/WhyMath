# S29R 추출기 — S28R 플래그 1 실행(f1) 기록에서 P2·P3·P4 × LF-01~05 × 1회차 15건을 골라 세션에 넘길 글로 만든다(읽기 전용)
# 사용: python extract_s29r.txt <f1 실행 폴더> <출력 txt>   (cwd = 작업 폴더의 src/backend · GOLDEN_SIM_SEED_DIR 설정)
# 고르는 기준은 셋뿐(페르소나 · 문항 · 1회차) — 내용으로 고르지 않는다(지시서 S24 할 일 2 · S29R 그대로 · 1회차 S29 추출기와 같은 규칙)
# 1회차 S29 추출기에서 바꾼 것: T2 매칭 줄 대신 T2R 채점 줄 — 학생 입력마다 실행 때와 같은 채점 함수
# (golden.grade.is_correct)를 플래그 GOLDEN_T2R_GRADE 0·1로 다시 불러 두 판정과 마지막 수식 조각을 붙인다
import json
import os
import sys
from pathlib import Path

# 파일로 실행하면 파이썬은 지금 폴더가 아니라 이 파일의 폴더를 먼저 본다 — 그러면 .venv에 연결된
# 공유 작업 사본의 golden이 잡힌다(S28R 블록 X 실측 · LATER 2026-11-27). 지금 폴더를 맨 앞에 둔다.
sys.path.insert(0, str(Path.cwd()))

import golden  # noqa: E402
from golden.grade import is_correct  # noqa: E402
from golden.grade_tail import FLAG_ENV, last_math_fragment  # noqa: E402
from golden.problems import load_problems  # noqa: E402
from golden.sim.metrics import expected_correct  # noqa: E402
from golden.sim.student import copies_seed, load_seed_utterances  # noqa: E402
from golden.tutor import FAKE_PREFIX  # noqa: E402

# 실행 코드 출처 — 지금 폴더(작업 폴더의 src/backend)의 golden이어야 한다(다른 사본 코드가 돌면 거부)
if not Path(golden.__file__).resolve().is_relative_to(Path.cwd().resolve()):
    sys.exit(f"S29R_REFUSED — golden이 다른 곳에서 왔다: {golden.__file__}")
print(f"GOLDEN_FROM={Path(golden.__file__).resolve().parent}")

PERSONAS = ("P2", "P3", "P4")
PROBLEMS = ("LF-01", "LF-02", "LF-03", "LF-04", "LF-05")
BANK = {p["id"]: p for p in load_problems()}


def grade(problem_id: str, text: str, flag: str) -> bool:
    """실행 때와 같은 채점 함수를 정해 둔 플래그로 부른다."""
    os.environ[FLAG_ENV] = flag
    return is_correct(BANK[problem_id], text)


run_dir, out_path = Path(sys.argv[1]), Path(sys.argv[2])
meta = json.loads((run_dir / "run_meta.json").read_text(encoding="utf-8"))
rows = [
    json.loads(ln)
    for ln in (run_dir / "sessions.jsonl").read_text(encoding="utf-8").splitlines()
    if ln.strip()
]
seeds = load_seed_utterances()

problems: list[str] = []
if meta.get("mode") != "live":
    problems.append(f"mode={meta.get('mode')}(live여야 한다)")

# 플래그 검증 — 이 기록이 정말 플래그 1 실행인가: 기록 전체의 학생 입력을 플래그 1로 다시 채점해 기록된 판정과 대조한다
# (run_meta에 플래그 칸이 없다 · S28R 추출기와 같은 방법)
flag_mismatch = 0
for r in rows:
    pid = str(r.get("problem_id"))
    for t in r.get("turns") or []:
        if bool(t.get("correct")) != grade(pid, str(t.get("student", "")), "1"):
            flag_mismatch += 1
if flag_mismatch:
    problems.append(f"기록 판정 ≠ 플래그 1 재채점 {flag_mismatch}턴(f1 실행 기록이 아니다)")

# 회차 = 그 페르소나의 몇 번째 블록인가(러너는 회차 → 페르소나 → 문항 순으로 한 줄씩 쓴다)
blocks: dict[str, int] = {}
prev = None
picked: dict[tuple[str, str], dict] = {}
for row in rows:
    p = row.get("persona")
    if p != prev:
        blocks[p] = blocks.get(p, 0) + 1
        prev = p
    if p in PERSONAS and blocks[p] == 1 and row.get("problem_id") in PROBLEMS:
        key = (p, row["problem_id"])
        if key in picked:
            sys.exit(f"S29R_REFUSED — 1회차에 같은 문항이 두 번 나왔다: {key}")
        picked[key] = row

missing = [f"{p}/{q}" for p in PERSONAS for q in PROBLEMS if (p, q) not in picked]
if missing:
    problems.append(f"빠진 조합 {missing}")
bad_http = [
    f"{k[0]}/{k[1]}={r.get('http_status')}"
    for k, r in picked.items()
    if not (isinstance(r.get("http_status"), int) and 200 <= r["http_status"] < 300)
]
if bad_http:
    problems.append(f"http_status 2xx 아님 {bad_http}")
fake = [
    f"{k[0]}/{k[1]}"
    for k, r in picked.items()
    if any(
        str(t.get("tutor", "")).startswith(FAKE_PREFIX)
        for t in r.get("turns", [])
        if t.get("kind") == "tutor_turn"
    )
]
if fake:
    problems.append(f"가짜 모드 튜터 발화 {fake}")

# 숫자 칸(세션이 손으로 다시 세어 대조한다)
# inputs 학생 입력 수 · wrong_f1 플래그 1 오답(튜터 턴 + 오답 판정) · flips 플래그 0·1 판정이 다른 입력 ·
# flips_known_wrong 그중 시뮬레이터가 오답으로 아는 입력 · sess_correct 정답으로 끝난 세션
num = {"inputs": 0, "wrong_f1": 0, "flips": 0, "flips_known_wrong": 0, "sess_correct": 0}
masked = 0
lines: list[str] = []
for p in PERSONAS:
    for q in PROBLEMS:
        r = picked.get((p, q))
        if r is None:
            continue
        turns = r.get("turns", [])
        last = turns[-1] if turns else {}
        forced = r.get("forced_answer")
        lines.append(
            f"## {p} · {q} · injected_op={r.get('injected_op')} · elapsed_s={r.get('elapsed_s')} · "
            f"http={r.get('http_status')} · 턴수={len(turns)} · 마지막={last.get('kind')} · correct={last.get('correct')}"
        )
        if last.get("kind") == "graded" and last.get("correct") is True:
            num["sess_correct"] += 1
        for t in turns:
            raw = str(t.get("student", ""))
            shown = raw
            if copies_seed(raw, seeds):
                shown = "[시드 일치 — 가림]"
                masked += 1
            kind = t.get("kind")
            g0 = grade(q, raw, "0")
            g1 = grade(q, raw, "1")
            num["inputs"] += 1
            num["wrong_f1"] += not g1
            exp = expected_correct(p, int(t.get("n", 0)), forced)
            if g0 != g1:
                num["flips"] += 1
                num["flips_known_wrong"] += exp is False and g1
            lines.append(f"[{t.get('n')}·{kind}] 학생: {shown}")
            # 마지막 수식 조각은 가린 입력이면 내보내지 않는다(시드 문장 조각이 될 수 있다)
            frag = "-" if shown != raw else (last_math_fragment(raw) or "없음")
            lines.append(
                f"[{t.get('n')}·{kind}] T2R: 플래그0={g0} · 플래그1={g1} · 뒤집힘={'예' if g0 != g1 else '아니오'} · "
                f"마지막 조각={frag} · 시뮬레이터 앎={exp}"
            )
            lines.append(f"[{t.get('n')}·{kind}] 튜터: {t.get('tutor', '')}")
        lines.append("")
os.environ.pop(FLAG_ENV, None)

ok = not problems
head = (
    f"S29R_CHECK ok={ok} picked={len(picked)} masked={masked} mode={meta.get('mode')} "
    f"rows={len(rows)} skipped={meta.get('skipped_sessions')} flag_mismatch={flag_mismatch}"
)
if problems:
    head += " problems=" + " | ".join(problems)
nums = "S29R_NUM " + " ".join(f"{k}={v}" for k, v in num.items())
text = head + "\n" + nums + "\n\n" + "\n".join(lines)
out_path.write_text(text, encoding="utf-8")
print(head)
print(nums)
print(f"OUT_FILE={out_path} CHARS={len(text)}")
sys.exit(0 if ok else 1)
